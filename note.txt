fn first_word(s: &String) -> usize { // return a byte index val
    let bytes = s.as_bytes(); // convert string to an arr of bytes 
    for(i, &item) in bytes.iter().enumerate() { // iter creates an iterator over the arr of bytes
        if item == b' ' { // byte literal
            return i;
        }
    }
    s.len()
}
